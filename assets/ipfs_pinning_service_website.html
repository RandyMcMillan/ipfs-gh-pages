import { PinPointClient } from '@pinpoint/ipfs-sdk';

const client = new PinPointClient({ 
  apiKey: process.env.PINPOINT_KEY 
});

async function pinContent() {
  const response = await client.pins.add({
    cid: 'bafybeigdyrzt5sfp7udm7hu76uh7y26nf3efuylqabf3oclgtqy55fbzdi',
    name: 'My Decentralized Asset'
  });
  
  console.log('Pinned Successfully Status:', response.status);
}
use pinpoint_ipfs_sdk::Client;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let client = Client::new("YOUR_PINPOINT_API_KEY");
    
    let pin = client.pin_cid("bafybeigdyrzt5sfp7udm7hu76uh7y26nf3efuylqabf3oclgtqy55fbzdi")
        .name("My Decentralized Asset")
        .send()
        .await?;

    println!("Pin Created CID: {}", pin.cid);
    Ok(())
}
